cmake_minimum_required(VERSION 3.16)
project(TestBootstrap)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
find_package(LibArchive REQUIRED)
add_executable(test_bootstrap 
    test_bootstrap_log.cpp
    src/runtime_bootstrap.cpp
    src/archive.cpp
    src/downloader.cpp
    src/download_dialog.cpp
)
target_include_directories(test_bootstrap PRIVATE ${CMAKE_SOURCE_DIR}/src)
target_link_libraries(test_bootstrap Qt6::Core Qt6::Widgets Qt6::Network LibArchive::LibArchive)
